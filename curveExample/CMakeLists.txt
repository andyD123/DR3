option(DR3_CURVE_SANITIZERS "Enable address/undefined-behaviour sanitizers on curve targets" OFF)
if(DEFINED DR3_ISA)
    set(_curve_isa_default "${DR3_ISA}")
else()
    set(_curve_isa_default AVX2)
endif()
set(DR3_CURVE_ISA "${_curve_isa_default}" CACHE STRING "SSE2, AVX2 or AVX512 for the curve targets")
set_property(CACHE DR3_CURVE_ISA PROPERTY STRINGS SSE2 AVX2 AVX512)
if(NOT DR3_CURVE_ISA MATCHES "^(SSE2|AVX2|AVX512)$")
    message(FATAL_ERROR "DR3_CURVE_ISA must be SSE2, AVX2 or AVX512")
endif()
if(DR3_ENABLE_TSAN AND (DR3_CURVE_SANITIZERS OR DR3_ENABLE_SANITIZERS))
    message(FATAL_ERROR "Curve ASan/UBSan cannot be combined with ThreadSanitizer")
endif()
if(DR3_BUILD_TESTS)
    enable_testing()
endif()

function(curve_target target vectorised)
    target_compile_features(${target} PUBLIC cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /EHsc)
    endif()
    if(${vectorised})
        if(DR3_CURVE_ISA STREQUAL "AVX512")
            target_compile_definitions(${target} PUBLIC DR3_CURVE_AVX512)
            if(MSVC)
                target_compile_options(${target} PUBLIC /arch:AVX512)
            else()
                target_compile_options(${target} PUBLIC -mavx512f -mfma)
            endif()
        elseif(DR3_CURVE_ISA STREQUAL "SSE2")
            target_compile_definitions(${target} PUBLIC DR3_CURVE_SSE2)
            if(NOT MSVC)
                target_compile_options(${target} PUBLIC -msse2)
            endif()
        elseif(MSVC)
            target_compile_options(${target} PUBLIC /arch:AVX2)
        else()
            target_compile_options(${target} PUBLIC -mavx2 -mfma)
        endif()
    endif()
    if(NOT MSVC)
        if(DR3_CURVE_SANITIZERS OR DR3_ENABLE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        elseif(DR3_ENABLE_TSAN)
            target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=thread)
        endif()
    endif()
endfunction()

if(DR3_BUILD_TESTS)
    foreach(target IN ITEMS curve_core_tests curve_edge_tests)
        add_executable(${target} ${target}.cpp)
        curve_target(${target} FALSE)
        add_test(NAME ${target} COMMAND ${target})

if(NOT DR3_CURVE_CORE_ONLY AND (DR3_BUILD_TESTS OR DR3_BUILD_EXAMPLES))
    find_package(Threads REQUIRED)
    # A target-scoped runtime avoids inheriting the historical root build's
    # -march=native setting; AVX2 here really means an AVX2-targeted build.
    add_library(dr3_curve_runtime STATIC
        ../Vectorisation/VecX/alloc_policy.cpp
        ../Vectorisation/VecX/vec.cpp)
    target_link_libraries(dr3_curve_runtime PUBLIC Threads::Threads)
    curve_target(dr3_curve_runtime TRUE)

    if(DR3_BUILD_TESTS)
        add_executable(curve_vector_tests curve_vector_tests.cpp)
        target_link_libraries(curve_vector_tests PRIVATE dr3_curve_runtime)
        curve_target(curve_vector_tests TRUE)
        add_test(NAME curve_vector_tests COMMAND curve_vector_tests)
    endif()
    if(DR3_BUILD_EXAMPLES)
        add_executable(curveExample curveExample.cpp)
        target_link_libraries(curveExample PRIVATE dr3_curve_runtime)
        curve_target(curveExample TRUE)
        if(DR3_BUILD_TESTS)
            add_test(NAME curve_demo_smoke COMMAND curveExample --bonds 8 --scenarios 9 --repeats 1)
            add_test(NAME curve_bad_scenario_count COMMAND curveExample --scenarios 0)
            set_tests_properties(curve_bad_scenario_count PROPERTIES WILL_FAIL TRUE)
        endif()
    endif()

    if(DR3_CURVE_ENABLE_QUANTLIB)
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(QL QUIET IMPORTED_TARGET quantlib)
        endif()
        if(TARGET PkgConfig::QL)
            set(CURVE_QUANTLIB_TARGET PkgConfig::QL)
        else()
            find_path(CURVE_QL_INCLUDE ql/version.hpp REQUIRED)
            find_library(CURVE_QL_LIBRARY NAMES QuantLib REQUIRED)
            add_library(curve_quantlib UNKNOWN IMPORTED)
            set_target_properties(curve_quantlib PROPERTIES
                IMPORTED_LOCATION "${CURVE_QL_LIBRARY}"
                INTERFACE_INCLUDE_DIRECTORIES "${CURVE_QL_INCLUDE}")
            set(CURVE_QUANTLIB_TARGET curve_quantlib)
        endif()
        if(DR3_BUILD_TESTS)
            add_executable(quantlib_adapter_tests quantlib_adapter_tests.cpp)
            target_link_libraries(quantlib_adapter_tests PRIVATE dr3_curve_runtime ${CURVE_QUANTLIB_TARGET})
            curve_target(quantlib_adapter_tests TRUE)
            add_test(NAME quantlib_adapter_tests COMMAND quantlib_adapter_tests)
        endif()
        if(DR3_BUILD_EXAMPLES)
            add_executable(quantlib_curve_demo quantlib_curve_demo.cpp)
            target_link_libraries(quantlib_curve_demo PRIVATE dr3_curve_runtime ${CURVE_QUANTLIB_TARGET})
            curve_target(quantlib_curve_demo TRUE)
            if(DR3_BUILD_TESTS)
                add_test(NAME quantlib_curve_demo_smoke COMMAND quantlib_curve_demo --bonds 8 --scenarios 9 --repeats 1)
            endif()
        endif()
    endif()
endif()
